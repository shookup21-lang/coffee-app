rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 날짜별 주문: 실제 "2026-09-22" 또는 테스트 "TEST-2026-09-22" 형식만 허용
    match /daily/{day} {
      allow read: if true;
      allow create, update: if day.matches('^(TEST-)?[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && (!('orders' in request.resource.data) || request.resource.data.orders.size() <= 300);
      allow delete: if false;
    }

    // 앱 설정: 메뉴·팀·주문 시간·공지·휴무일 5개 문서만 허용
    match /config/{name} {
      allow read: if true;
      allow create, update: if name in ['menu', 'teams', 'hours', 'notice', 'holidays'];
      allow delete: if false;
    }

    // 그 밖의 모든 경로는 규칙이 없으므로 자동으로 막힘
  }
}
